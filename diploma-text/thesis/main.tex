% --- Стиль листингов для текстовых промптов (приложение В) ---
% Базовый codestyle из 20-listing.sty имеет опечатку extendedchars=\true,
% из-за которой кириллица и спецсимволы рендерятся некорректно. Здесь
% задан собственный стиль без подсветки ключевых слов/строк/комментариев
% и с корректным extendedchars=true.
\lstdefinestyle{promptstyle}{
    frame=single,
    basicstyle=\ttfamily\footnotesize,
    numbers=none,
    breakatwhitespace=false,
    showstringspaces=false,
    breaklines=true,
    keepspaces=true,
    extendedchars=true,
    columns=fullflexible,
    upquote=true,
    backgroundcolor=\color{backcolour},
}
